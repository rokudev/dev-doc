export const configMap = {
    info: {
        role: "status",
        ariaLive: "polite",
        label: "Information",
        backgroundColor: "bg-[#e5f4ff]",
        border: "border-[#a9dbff]",

    },

export const Alert = ({ type = "info", title, children, closable = true, className = "", }) => {
    const [isVisible, setIsVisible] = useState(true);
    if (!isVisible)
        return null;
    const config = configMap[type] || configMap.info;
    return (<div className={`flex items-start gap-3 pt-3.5 px-4.5 mb-4 border rounded-[4px] text-left transition-all box-border ${config.backgroundColor} ${config.border} ${config.text} ${className}`} role={config.role} aria-live={config.ariaLive} aria-atomic="true">
      {/* Left Icon */}
      <div className="flex items-center justify-center w-3.5 h-3.5 mt-0.5 shrink-0 text-current" aria-hidden="true">
        {config.icon}
      </div>

      {/* Content Block (Title & Subtitle) */}
      <div className="flex-1 flex flex-col gap-1 min-w-0">
        <div className="text-sm font-bold leading-tight text-current">
          {title}
        </div>
        {children && (<div className="text-[13px] leading-snug text-current [&_p]:m-0 [&_p]:p-0">
            {children}
          </div>)}
      </div>

      {/* Close / Dismiss Button */}
      {closable && (<button type="button" onClick={() => setIsVisible(false)} className="flex items-center justify-center w-[18px] h-[18px] p-0 m-0 mt-px shrink-0 bg-transparent border-0 text-current hover:opacity-70 transition-opacity rounded-sm focus:outline-none focus-visible:ring-2 focus-visible:ring-current" aria-label={`Dismiss ${config.label} alert`}>
          <DismissIcon />
        </button>)}
    </div>);
};
