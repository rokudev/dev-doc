import React from "react";
export const BlockQuote = ({ children, className = "", textAlign = "left", verticalAlign = "center", }) => {
    const alignmentClasses = {
        text: {
            left: "text-left",
            center: "text-center",
            right: "text-right",
        },
        vertical: {
            top: "items-start",
            center: "items-center",
            bottom: "items-end",
        },
    };
    return (<div className={`rm-blockquote-bg border border-solid rm-blockquote-border dark:bg-yellow-900 p-5 flex ${alignmentClasses.vertical[verticalAlign]} ${className}`}>
        {children}
    </div>);
};


{/* Example Usage */}
<BlockQuote>
  <div>ArrayGrid is not meant to be instantiated directly by app code</div>
</BlockQuote>
