export const SimpleTable = ({header, columns, data}) => {
  const containerStyle = { 
    maxWidth: '600px', 		
    margin: '40px auto', 
    padding: '10px', 
    border: '1px solid #ccc', 
    borderRadius: '2px', 
    textAlign: 'center'
  };

  return (
    <div style={containerStyle}>
      
			<h2>{header}</h2>

      <table style={{ borderCollapse: 'collapse', width: '100%' }}>
       	<thead>
          <tr>
            {columns.map((col) => (
              <th key={col} style={{ border: '1px solid #ddd', padding: '8px', color: 'black' }}>
                {col}
              </th>
            ))}
          </tr>
        </thead>
        
				<tbody>
          {data.map((row, rowIndex) => (
            <tr key={rowIndex}>
              {columns.map((col) => (
                <td key={col} style={{ border: '1px solid #ddd', padding: '8px', color: 'black' }}>
                  {row[`${col}-pre`] && <div>{row[`${col}-pre`]}<br/></div>}
                  {row[col]}
									{row[`${col}-post`] && <div>{row[`${col}-post`]}</div>}
                </td>
              ))}
            </tr>
          ))}
      	</tbody>
      </table>
	</div>

    );
};

// To use this in your page, copy the next line and paste into your page and customize the list of items.
<SimpleTable header="Simple Table" columns={['Name', 'Age', 'Country']} data={[
    { Name: 'Alice', Age: 25, Country: 'USA' },
    { Name: 'Bob', Age: 30, 'Country-pre': 'I am from ', Country: 'UK.', 'Country-post': 'Hello everyone' },
    { Name: 'Charlie', Age: 28, Country: 'Canada' }
  ]} />
