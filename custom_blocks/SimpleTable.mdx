export const SimpleTable = ({columns, data}) => {
    return (
      <table style={{ borderCollapse: 'collapse', width: '100%' }}>
       	<thead>
          <tr>
            {columns.map((col) => (
              <th key={col} style={{ border: '1px solid #ddd', padding: '8px', color: 'black' }}>
                {col}
              </th>
            ))}
          </tr>
        </thead>
        
				<tbody>
          {data.map((row, rowIndex) => (
            <tr key={rowIndex}>
              {columns.map((col) => (
                <td key={col} style={{ border: '1px solid #ddd', padding: '8px', color: 'black' }}>
                  {row[col]}
                </td>
              ))}
            </tr>
          ))}
      	</tbody>
       </table>
    );
};

// To use this in your page, copy the next line and paste into your page and customize the list of items.
<SimpleTable columns={['Name', 'Age', 'Country']} data={[
    { Name: 'Alice', Age: 25, Country: 'USA' },
    { Name: 'Bob', Age: 30, Country: 'UK' },
    { Name: 'Charlie', Age: 28, Country: 'Canada' }
  ]} />
