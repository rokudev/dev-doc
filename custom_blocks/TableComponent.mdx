export const TableComponent = ({ header }) => {
  const containerStyle = { 
    maxWidth: '800px', 		
    margin: '5px auto',
    padding: '0 5px 5px 5px', 
    border: '1px solid #ccc', 
    borderRadius: '8px', 
    boxShadow: 'rgba(0, 0, 0, 0.35) 0px 5px 15px',
    backgroundColor: 'oklch(87.2% 0.01 258.338)',
    textAlign: 'left'
  };

  const tableStyle = {
    width: '100%',
    borderCollapse: 'collapse',
    //marginTop: '20px',
    backgroundColor: 'white',
    borderRadius: '4px',
    overflow: 'hidden'
  };

  const headerStyle = {
    backgroundColor: '#f5f5f5',
    fontWeight: 'bold',
    padding: '12px',
    border: '1px solid #ddd',
    textAlign: 'left'
  };
  const cellStyle = {
    padding: '12px',
    border: '1px solid #ddd',
    verticalAlign: 'top'
  };

  const listStyle = {
    margin: '8px 0',
    paddingLeft: '20px'
  };

  const listItemStyle = {
    marginBottom: '4px'
  };
  const renderCellContent = (value, type) => {
    if (value === null || value === undefined) return '';

    if (type === 'permissions' || (Array.isArray(value) && typeof value[0] === 'object')) {
      return (
        <ul style={listStyle}>
          {value.map((item, index) => (
            <li key={index} style={listItemStyle}>
              {item.category && <strong>{item.category}:</strong>} {item.text}
              {item.subItems && item.subItems.length > 0 && (
                <ul style={listStyle}>
                  {item.subItems.map((subItem, subIndex) => (
                    <li key={subIndex} style={listItemStyle}>
                      {subItem}
                    </li>
                  ))}
                </ul>
              )}
            </li>
          ))}
        </ul>
      );
    }

    if (Array.isArray(value)) {
      return (
        <ul style={listStyle}>
          {value.map((item, index) => (
            <li key={index} style={listItemStyle}>{item}</li>
          ))}
        </ul>
      );
    }

    if (typeof value === 'object') {
      return JSON.stringify(value);
    }

    return String(value);
  };


  return (
    <div style={containerStyle}>
      <h2>{header}</h2>
      
			<table style={tableStyle}>
        <thead>
          <tr>
            {columns ? columns.map((column, index) => (
              <th key={index} style={headerStyle}>{column.label}</th>
            )) : Object.keys(data[0] || {}).map((key, index) => (
              <th key={index} style={headerStyle}>{key.charAt(0).toUpperCase() + key.slice(1)}</th>
            ))}
          </tr>
        </thead>
        
        <tbody>
          {data.map((row, index) => (
            <tr key={index}>
              {columns ? columns.map((column, colIndex) => (
                <td key={colIndex} style={cellStyle}>
                  {renderCellContent(row[column.key], column.type)}
                </td>
              )) : Object.entries(row).map(([key, value], colIndex) => (
                <td key={colIndex} style={cellStyle}>
                  {renderCellContent(value)}
                </td>
              ))}
            </tr>
          ))}
        </tbody>
			</table>
		</div>
  );
};
